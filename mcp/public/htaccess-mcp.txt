# ---- Claude-Connector (MCP) – Ergänzung für public/.htaccess (vor den bestehenden Regeln einfügen) ----
RewriteEngine On

# Authorization-Kopf (Bearer-Token) an PHP weitergeben – IONOS reicht ihn sonst nicht durch
RewriteCond %{HTTP:Authorization} .
RewriteRule .* - [E=HTTP_AUTHORIZATION:%{HTTP:Authorization}]

# Wege des Connectors
RewriteRule ^mcp/?$ mcp.php [L,QSA]
RewriteRule ^oauth/(register|authorize|token|revoke)$ oauth.php?weg=/oauth/$1 [L,QSA]
# IONOS: für /.well-known/ nur mit absolutem Ziel (relatives Ziel ergibt 404)
RewriteCond %{REQUEST_URI} ^/\.well-known/(oauth-protected-resource|oauth-authorization-server|openid-configuration)
RewriteRule ^ /oauth.php [L]
